# Augmented Dickey-Fuller Test for stationarity on India CPI time series
print("H0 : The time series is non-stationary")
print("H1 : The time series is stationary")
print("alpha = 0.05")
print("A p-value less than 0.05 indicates that we can reject H0 \n")

result = adfuller(df["India"].dropna())

print("Augmented Dickey-Fuller Test:")
labels = ['ADF Statistic', 'p-value', 'Lags Used', 'Number of Observations Used']

for value, label in zip(result, labels):
        print(f'{label}: {value}')

if result[1] <= 0.05:
        print("\nReject the null hypothesis => The time series is stationary")
else:
        print("\nFail to reject the null hypothesis => The time series is non-stationary")     
# The p-value is greater than 0.05, we fail to reject H0, 
# indicating that the time series is non-stationary.
